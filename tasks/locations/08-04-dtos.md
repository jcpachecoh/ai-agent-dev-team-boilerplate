## Implementation notes

### `UserLocationType` (respuesta de `assignUserToLocation`)

```typescript
@ObjectType('UserLocation')
export class UserLocationType {
  @Field(() => ID) id!: string;
  @Field(() => ID) userId!: string;
  @Field(() => ID) locationId!: string;
  @Field() restaurantId!: string; // desde tenantId
  @Field() createdAt!: Date;
}

export function toUserLocationType(row: UserLocation): UserLocationType {
  return {
    id: row.id,
    userId: row.userId,
    locationId: row.locationId,
    restaurantId: row.tenantId,
    createdAt: row.createdAt,
  };
}
```

### `LocationAssignedUserType` (respuesta de `locationUsers`)

> Nombre GraphQL `LocationAssignedUser` — evita colisión con `User` de `user.resolver.ts` (OperationalUser).

```typescript
@ObjectType('LocationAssignedUser')
export class LocationAssignedUserType {
  @Field(() => ID) id!: string;
  @Field() email!: string;
  @Field() cedula!: string;
  @Field() isActive!: boolean;
  @Field() restaurantId!: string; // desde user.tenantId
}

export function toLocationAssignedUserType(user: User): LocationAssignedUserType {
  return {
    id: user.id,
    email: user.email,
    cedula: user.cedula,
    isActive: user.isActive,
    restaurantId: user.tenantId,
  };
}
```
